  /** Run a trusted local project-session operation through normal delegation events. */
  async runManagedProjectSession(action, input, {
    ownerId,
    coordinationRunId,
    cwd,
  } = {}) {
    const runId = clean(coordinationRunId)
    const run = this.managedCoordinatorRuns.get(runId)
    if (!run || run.ownerId !== clean(ownerId)) {
      throw new AgentError(
        'COORDINATOR_RUN_UNAVAILABLE: the active Coordinator turn cannot accept this delegation',
        { status: 409, protocol: this.protocol },
      )
    }
    if (run.delegation) {
      throw new AgentError(
        'COORDINATOR_ALREADY_DELEGATED: this Coordinator turn already started a project Session',
        { status: 409, protocol: this.protocol },
      )
    }
    run.cwd = clean(cwd) || run.cwd || this.directory
    const started = action === 'start'
      ? await this.startProjectSession(run, input)
      : await this.continueProjectSession(run, input)
    const record = run.delegation
    if (!record?.promise) {
      throw new AgentError(
        'DELEGATION_NOT_STARTED: DSH did not create a background Session run',
        { status: 502, protocol: this.protocol },
      )
    }
    // Do not await record.promise here. runCoordinator() owns that promise,
    // finalizes the durable Coordinator turn, and emits completion events.
    return {
      content: '',
      metadata: { delegation: started },
    }
  }
